PS C:\WINDOWS\system32> param(
    [Parameter(Mandatory = $true, Position = 0, HelpMessage = "Enter the root folder path")]
    [string]$Path,

    [string]$LogFile = "FolderHierarchy.log"
)

function Show-FolderHierarchy {
    param(
        [string]$CurrentPath,
        [string]$Indent = ""
    )

    $dirs = Get-ChildItem -Path $CurrentPath -Directory | Sort-Object Name

    for ($i = 0; $i -lt $dirs.Count; $i++) {
        $dir = $dirs[$i]
        $isLast = ($i -eq ($dirs.Count - 1))
        $connector = if ($isLast) { "└── " } else { "├── " }
        $line = "$Indent$connector$($dir.Name)"

        Write-Output $line
        if ($LogFile) {
            Add-Content -Path $LogFile -Value $line -Encoding UTF8
        }

        $newIndent = $Indent + (if ($isLast) { "    " } else { "│   " })
        Show-FolderHierarchy -CurrentPath $dir.FullName -Indent $newIndent
    }
}

# Clear previous log
if (Test-Path $LogFile) { Remove-Item $LogFile -Force }

Write-Output "Folder Structure for: $Path"
"Folder Structure for: $Path`n" | Out-File -FilePath $LogFile -Encoding UTF8

if (-not (Test-Path $Path -PathType Container)) {
    Write-Error "❌ Invalid folder path: $Path"
} else {
    Show-FolderHierarchy -CurrentPath $Path
    Write-Output "`n✅ Hierarchy logged to: $LogFile"
}
